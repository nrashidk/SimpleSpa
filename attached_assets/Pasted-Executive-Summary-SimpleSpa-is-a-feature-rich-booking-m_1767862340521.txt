SimpleSpa is a feature-rich booking management system with solid foundations in authentication, encryption, and audit logging. However, critical security middleware is not configured despite being installed, and several high-impact vulnerabilities create immediate exploit vectors. The 5,486-line routes file and monolithic architecture present significant maintainability risks that will impede future development.

Immediate Action Required:

Configure helmet, CORS, and rate limiting middleware
Remove/protect development endpoints exposed in production
Implement CSRF protection
Refactor monolithic routes.ts (5,486 lines)

    return res.status(error.status).json({ 
      message: error.message, 
      code: error.code 
    });
  }

    stack: error.stack,
    code: error.code,
    timestamp: new Date().toISOString()
  });

  if (process.env.NODE_ENV === 'production') {
    return res.status(500).json({ 
      message: "An unexpected error occurred. Please try again later.",
      errorId: generateErrorId() // Track errors without exposing details
    });
  }
  
  // Development: Include details
  res.status(500).json({ 
    message: error.message || message,
    stack: error.stack,
    code: error.code
  });
}

import crypto from 'crypto';

function signOAuthState(data: object): string {
  const payload = JSON.stringify(data);
  const signature = crypto
    .createHmac('sha256', process.env.OAUTH_STATE_SECRET!)
    .update(payload)
    .digest('hex');
  return Buffer.from(JSON.stringify({ payload, signature })).toString('base64');
}

function verifyOAuthState(state: string): object {
  const { payload, signature } = JSON.parse(Buffer.from(state, 'base64').toString());
  const expectedSignature = crypto
    .createHmac('sha256', process.env.OAUTH_STATE_SECRET!)
    .update(payload)
    .digest('hex');
  if (signature !== expectedSignature) {
    throw new Error('Invalid state signature');
  }
  return JSON.parse(payload);
}

interface LoginAttempt {
  email: string;
  attempts: number;
  lockedUntil?: Date;
  lastAttempt: Date;
}

const loginAttempts = new Map<string, LoginAttempt>();

async function checkLoginAttempts(email: string): Promise<{ allowed: boolean; reason?: string }> {
  const attempt = loginAttempts.get(email);
  
  if (!attempt) {
    return { allowed: true };
  }
  
  // Check if account is locked
  if (attempt.lockedUntil && attempt.lockedUntil > new Date()) {
    const minutesLeft = Math.ceil((attempt.lockedUntil.getTime() - Date.now()) / 60000);
    return { 
      allowed: false, 
      reason: `Account locked for ${minutesLeft} more minutes due to multiple failed login attempts` 
    };
  }
  
  // Reset if last attempt was > 15 minutes ago
  if (Date.now() - attempt.lastAttempt.getTime() > 15 * 60 * 1000) {
    loginAttempts.delete(email);
    return { allowed: true };
  }
  
  return { allowed: true };
}

async function recordFailedLogin(email: string) {
  const attempt = loginAttempts.get(email) || { email, attempts: 0, lastAttempt: new Date() };
  attempt.attempts++;
  attempt.lastAttempt = new Date();
  
  // Progressive lockout: 5 fails = 15 min, 10 fails = 1 hour, 15 fails = 24 hours
  if (attempt.attempts >= 15) {
    attempt.lockedUntil = new Date(Date.now() + 24 * 60 * 60 * 1000);
  } else if (attempt.attempts >= 10) {
    attempt.lockedUntil = new Date(Date.now() + 60 * 60 * 1000);
  } else if (attempt.attempts >= 5) {
    attempt.lockedUntil = new Date(Date.now() + 15 * 60 * 1000);
  }
  
  loginAttempts.set(email, attempt);
  
  // Log suspicious activity
  await AuditLogger.log({
    action: 'FAILED_LOGIN',
    entityType: 'security',
    details: { email, attempts: attempt.attempts }
  });
}

// server/middleware/errorHandler.ts
export function errorHandler(err: Error, req: Request, res: Response, next: NextFunction) {
  if (err instanceof AppError) {
    return res.status(err.statusCode).json({
      error: {
        message: err.message,
        code: err.code,
        details: err.details
      }
    });
  }
  
  // Log unexpected errors
  console.error('Unexpected error:', err);
  
  res.status(500).json({
    error: {
      message: process.env.NODE_ENV === 'production' 
        ? 'An unexpected error occurred' 
        : err.message,
      code: 'INTERNAL_ERROR'
    }
  });
}
